    <script>
        /* =========================================================================
           GLOBAL KONFIGURATION & KONTROLLER
           ========================================================================= */
        
        let globalTimeScale = 1.0;
        let activeTab = "tab-bridge";

        // Undgå brug af dollar-tegn. Vi laver en simpel helper til at hente elementer.
        function getEl(id) {
            return document.getElementById(id);
        }

        // Håndtering af Tabs
        const tabBtns = document.querySelectorAll(".tab-btn");
        const tabContents = document.querySelectorAll(".tab-content");

        tabBtns.forEach(function(btn) {
            btn.addEventListener("click", function() {
                // Fjern active fra alle
                for(let i=0; i<tabBtns.length; i++) {
                    tabBtns[i].classList.remove("active");
                    tabContents[i].classList.remove("active");
                }
                // Sæt active på valgt
                btn.classList.add("active");
                const targetId = btn.getAttribute("data-target");
                getEl(targetId).classList.add("active");
                activeTab = targetId;
                
                // Tving resize af canvas
                window.dispatchEvent(new Event('resize'));
            });
        });

        // Håndtering af tids-hastighed
        const timeBtns = document.querySelectorAll(".time-btn");
        timeBtns.forEach(function(btn) {
            btn.addEventListener("click", function() {
                for(let i=0; i<timeBtns.length; i++) {
                    timeBtns[i].classList.remove("active");
                }
                btn.classList.add("active");
                // Konverter tekst-komma til punktum for JS logik
                globalTimeScale = parseFloat(btn.getAttribute("data-speed").replace(",", "."));
            });
        });

        // Hjælpefunktion til at formatere tal med dansk komma
        function formatDK(num, decimals) {
            if(decimals === undefined) decimals = 0;
            return num.toFixed(decimals).replace(".", ",");
        }


        /* =========================================================================
           GRAF-VISUALISERINGS MOTOR (Datavisualisering)
           ========================================================================= */
        class GraphEngine {
            constructor(canvasId, lineColors, yMaxInitial) {
                this.canvas = getEl(canvasId);
                this.ctx = this.canvas.getContext("2d");
                this.data = []; // Array of arrays: [ [y1, y2], [y1, y2] ... ]
                this.lineColors = lineColors;
                this.maxDataPoints = 150;
                this.yMax = yMaxInitial;
            }

            resize() {
                const rect = this.canvas.parentElement.getBoundingClientRect();
                this.canvas.width = rect.width;
                this.canvas.height = rect.height;
            }

            addData(dataPoints) {
                this.data.push(dataPoints);
                if (this.data.length > this.maxDataPoints) {
                    this.data.shift();
                }
            }

            draw() {
                const ctx = this.ctx;
                const width = this.canvas.width;
                const height = this.canvas.height;

                ctx.clearRect(0, 0, width, height);

                // Tegn baggrunds-gitter
                ctx.strokeStyle = "rgba(255, 255, 255, 0.1)";
                ctx.lineWidth = 1;
                ctx.beginPath();
                for (let i = 1; i < 4; i++) {
                    let y = height - (i * (height / 4));
                    ctx.moveTo(0, y);
                    ctx.lineTo(width, y);
                }
                ctx.stroke();

                if (this.data.length < 2) return;

                // Dynamisk justering af y-akse hvis værdier overstiger yMax
                let currentMax = 0;
                for (let i = 0; i < this.data.length; i++) {
                    for (let j = 0; j < this.data[i].length; j++) {
                        if (this.data[i][j] > currentMax) currentMax = this.data[i][j];
                    }
                }
                if (currentMax > this.yMax * 0.9) {
                    this.yMax = currentMax * 1.2; // Zoom ud
                } else if (currentMax < this.yMax * 0.4 && this.yMax > 100) {
                    this.yMax = this.yMax * 0.99; // Langsom zoom ind
                }

                // Tegn data linjer
                const stepX = width / (this.maxDataPoints - 1);

                for (let lineIdx = 0; lineIdx < this.lineColors.length; lineIdx++) {
                    ctx.strokeStyle = this.lineColors[lineIdx];
                    ctx.lineWidth = 3;
                    ctx.beginPath();
                    
                    for (let i = 0; i < this.data.length; i++) {
                        let val = this.data[i][lineIdx];
                        let x = i * stepX;
                        let y = height - (val / this.yMax) * (height - 10) - 5;
                        
                        if (i === 0) ctx.moveTo(x, y);
                        else ctx.lineTo(x, y);
                    }
                    ctx.stroke();
                }
            }
        }


        /* =========================================================================
           SIMULATION 1: LILLEBÆLTSBROERNE (DYNAMISK LIGEVÆGT)
           ========================================================================= */
        class Car {
            constructor(x, y, targetX, targetY, speed) {
                this.x = x;
                this.y = y;
                this.targetX = targetX;
                this.targetY = targetY;
                this.speed = speed;
                this.color = Math.random() > 0.5 ? "#e74c3c" : "#f1c40f";
                this.arrived = false;
            }

            update(dt) {
                let dx = this.targetX - this.x;
                let dy = this.targetY - this.y;
                let dist = Math.sqrt(dx*dx + dy*dy);

                if (dist < 5) {
                    this.arrived = true;
                } else {
                    let moveDist = this.speed * dt * globalTimeScale;
                    if (moveDist > dist) moveDist = dist;
                    this.x += (dx / dist) * moveDist;
                    this.y += (dy / dist) * moveDist;
                }
            }

            draw(ctx) {
                ctx.fillStyle = this.color;
                ctx.beginPath();
                ctx.arc(this.x, this.y, 4, 0, Math.PI * 2);
                ctx.fill();
            }
        }

        class BridgeSimulation {
            constructor() {
                this.canvas = getEl("canvas-bridge");
                this.ctx = this.canvas.getContext("2d");
                this.graph = new GraphEngine("graph-bridge", ["#3498db", "#e67e22"], 200);
                
                this.carsFyn = 100;
                this.carsJylland = 50;
                this.movingCars = [];

                // Konstanter for reaktionshastighed
                this.kFrem = 0.05; // Sandsynlighed for at en bil kører Fyn -> Jylland
                this.kTilbage = 0.05; // Sandsynlighed for Jylland -> Fyn
                
                this.timer = 0;
                this.resize();

                // Setup knapper
                getEl("btn-add-fyn").addEventListener("click", () => { this.carsFyn += 50; });
                getEl("btn-add-jylland").addEventListener("click", () => { this.carsJylland += 50; });
                getEl("btn-reset-bridge").addEventListener("click", () => {
                    this.carsFyn = 100;
                    this.carsJylland = 50;
                    this.movingCars = [];
                    this.graph.data = [];
                });
            }

            resize() {
                const rect = this.canvas.parentElement.getBoundingClientRect();
                this.canvas.width = rect.width;
                this.canvas.height = rect.height;
                this.graph.resize();
            }

            update(dt) {
                // Logik for dynamisk ligevægt
                // Antal biler der vil køre pr tidsenhed
                this.timer += dt * globalTimeScale;
                
                if (this.timer > 0.1) {
                    this.timer = 0;
                    
                    // Fyn -> Jylland (frem)
                    let carsMovingForward = Math.floor(this.carsFyn * this.kFrem * Math.random() * 2);
                    if (carsMovingForward > this.carsFyn) carsMovingForward = this.carsFyn;
                    
                    for(let i=0; i<carsMovingForward; i++) {
                        let startX = this.canvas.width * 0.2 + (Math.random() * 40 - 20);
                        let startY = this.canvas.height * 0.4 + (Math.random() * 20 - 10);
                        let endX = this.canvas.width * 0.8 + (Math.random() * 40 - 20);
                        let endY = this.canvas.height * 0.4 + (Math.random() * 20 - 10);
                        this.movingCars.push(new Car(startX, startY, endX, endY, 150));
                        this.carsFyn--;
                    }

                    // Jylland -> Fyn (tilbage)
                    let carsMovingBackward = Math.floor(this.carsJylland * this.kTilbage * Math.random() * 2);
                    if (carsMovingBackward > this.carsJylland) carsMovingBackward = this.carsJylland;

                    for(let i=0; i<carsMovingBackward; i++) {
                        let startX = this.canvas.width * 0.8 + (Math.random() * 40 - 20);
                        let startY = this.canvas.height * 0.6 + (Math.random() * 20 - 10);
                        let endX = this.canvas.width * 0.2 + (Math.random() * 40 - 20);
                        let endY = this.canvas.height * 0.6 + (Math.random() * 20 - 10);
                        this.movingCars.push(new Car(startX, startY, endX, endY, 150));
                        this.carsJylland--;
                    }

                    // Opdater graf
                    this.graph.addData([this.carsFyn, this.carsJylland]);
                }

                // Opdater biler i transit
                for(let i = this.movingCars.length - 1; i >= 0; i--) {
                    let car = this.movingCars[i];
                    car.update(dt);
                    if (car.arrived) {
                        if (car.targetX > this.canvas.width / 2) {
                            this.carsJylland++; // Ankommet til Jylland
                        } else {
                            this.carsFyn++; // Ankommet til Fyn
                        }
                        this.movingCars.splice(i, 1);
                    }
                }

                // UI opdatering
                getEl("stat-fyn").innerText = this.carsFyn;
                getEl("stat-jylland").innerText = this.carsJylland;
                
                let yVal = this.carsFyn > 0 ? (this.carsJylland / this.carsFyn) : 0;
                getEl("stat-bridge-y").innerText = formatDK(yVal, 2);
            }

            draw() {
                const ctx = this.ctx;
                const w = this.canvas.width;
                const h = this.canvas.height;

                ctx.clearRect(0, 0, w, h);

                // Tegn Landemasser
                ctx.fillStyle = "#27ae60";
                ctx.fillRect(0, 0, w * 0.3, h); // Fyn
                ctx.fillStyle = "#2ecc71";
                ctx.fillRect(w * 0.7, 0, w * 0.3, h); // Jylland

                // Vand (Lillebælt)
                ctx.fillStyle = "#2980b9";
                ctx.fillRect(w * 0.3, 0, w * 0.4, h);

                // Tekst for landemasser
                ctx.fillStyle = "rgba(255,255,255,0.7)";
                ctx.font = "bold 24px Arial";
                ctx.textAlign = "center";
                ctx.fillText("Fyn", w * 0.15, 40);
                ctx.fillText("(Middelfart)", w * 0.15, 70);
                
                ctx.fillText("Jylland", w * 0.85, 40);
                ctx.fillText("(Fredericia)", w * 0.85, 70);

                // Tegn Broer
                ctx.fillStyle = "#7f8c8d";
                // Gamle bro
                ctx.fillRect(w * 0.3, h * 0.35, w * 0.4, h * 0.1);
                // Nye bro
                ctx.fillRect(w * 0.3, h * 0.55, w * 0.4, h * 0.1);

                ctx.fillStyle = "white";
                ctx.font = "12px Arial";
                ctx.fillText("Den Gamle Lillebæltsbro ⟶", w * 0.5, h * 0.35 - 10);
                ctx.fillText("⟵ Den Nye Lillebæltsbro", w * 0.5, h * 0.55 - 10);

                // Visualiser koncentration (Biler i byerne)
                ctx.fillStyle = "rgba(0,0,0,0.4)";
                ctx.fillRect(w * 0.05, h * 0.2, w * 0.2, h * 0.6);
                ctx.fillRect(w * 0.75, h * 0.2, w * 0.2, h * 0.6);

                // Tegn biler der venter
                for(let i=0; i<Math.min(this.carsFyn, 200); i++) {
                    ctx.fillStyle = "#3498db";
                    ctx.fillRect(w * 0.06 + (i%10)*8, h * 0.21 + Math.floor(i/10)*8, 6, 6);
                }

                for(let i=0; i<Math.min(this.carsJylland, 200); i++) {
                    ctx.fillStyle = "#e67e22";
                    ctx.fillRect(w * 0.76 + (i%10)*8, h * 0.21 + Math.floor(i/10)*8, 6, 6);
                }

                // Tegn biler i transit
                for(let i=0; i<this.movingCars.length; i++) {
                    this.movingCars[i].draw(ctx);
                }

                this.graph.draw();
            }
        }


        /* =========================================================================
           SIMULATION 2: NATURENS ØKOBALANCE
           ========================================================================= */
        class NatureSimulation {
            constructor() {
                this.canvas = getEl("canvas-nature");
                this.ctx = this.canvas.getContext("2d");
                this.graph = new GraphEngine("graph-nature", ["#f1c40f", "#2ecc71"], 1500); // Gul=Rensdyr, Grøn=Føde
                
                this.reindeer = 200;
                this.food = 1000;
                
                this.carryingCapacityFood = 2500;
                this.foodGrowthRate = 25; // pr. tick
                this.reindeerEatRate = 0.05; // sandsynlighed for at spise pr. tick
                this.reindeerStarveRate = 0.01; // dør hvis føde mangler

                this.timer = 0;
                this.snowflakes = [];
                for(let i=0; i<50; i++) {
                    this.snowflakes.push({
                        x: Math.random(),
                        y: Math.random(),
                        speed: 0.1 + Math.random() * 0.3
                    });
                }

                this.resize();

                // Setup knapper
                getEl("btn-add-reindeer").addEventListener("click", () => { this.reindeer += 1000; });
                getEl("btn-add-food").addEventListener("click", () => { this.food += 500; });
                getEl("btn-reset-nature").addEventListener("click", () => {
                    this.reindeer = 200;
                    this.food = 1000;
                    this.graph.data = [];
                });
            }

            resize() {
                const rect = this.canvas.parentElement.getBoundingClientRect();
                this.canvas.width = rect.width;
                this.canvas.height = rect.height;
                this.graph.resize();
            }

            update(dt) {
                this.timer += dt * globalTimeScale;
                
                if (this.timer > 0.1) {
                    this.timer = 0;
                    
                    // Økologisk logik (forenklet Lotka-Volterra aktig)
                    // Føde vokser op til bæreevne
                    this.food += this.foodGrowthRate;
                    if (this.food > this.carryingCapacityFood) this.food = this.carryingCapacityFood;

                    // Rensdyr interagerer med føde
                    let neededFood = this.reindeer * this.reindeerEatRate;
                    
                    if (this.food >= neededFood && this.reindeer > 0) {
                        // Der er nok mad. Populationen stabiliseres eller vokser svagt
                        this.food -= neededFood;
                        // Formering hvis der er overskudsmad per rensdyr
                        if (this.food > this.reindeer * 2 && Math.random() > 0.5) {
                            this.reindeer += Math.floor(this.reindeer * 0.02);
                        }
                    } else {
                        // Mangel på mad -> sult
                        if (this.food > 0) {
                            this.food -= neededFood * 0.5; // Spiser resterne
                            if (this.food < 0) this.food = 0;
                        }
                        
                        let starved = Math.floor(this.reindeer * this.reindeerStarveRate * 5);
                        this.reindeer -= starved;
                        if (this.reindeer < 0) this.reindeer = 0;
                    }

                    this.graph.addData([this.reindeer, this.food]);
                }

                // Opdater snefnug
                for(let i=0; i<this.snowflakes.length; i++) {
                    let sf = this.snowflakes[i];
                    sf.y += sf.speed * dt * globalTimeScale;
                    if(sf.y > 1) {
                        sf.y = 0;
                        sf.x = Math.random();
                    }
                }

                // UI opdatering
                getEl("stat-reindeer").innerText = formatDK(this.reindeer);
                getEl("stat-food").innerText = formatDK(this.food);
            }

            draw() {
                const ctx = this.ctx;
                const w = this.canvas.width;
                const h = this.canvas.height;

                // Baggrundsskov
                ctx.fillStyle = "#2c3e50";
                ctx.fillRect(0, 0, w, h);
                
                // Jord
                ctx.fillStyle = "#34495e";
                ctx.fillRect(0, h * 0.6, w, h * 0.4);

                // Tegn "Mængden" af føde som grøn farve på jorden
                let foodRatio = this.food / this.carryingCapacityFood;
                ctx.fillStyle = "rgba(46, 204, 113, " + (foodRatio * 0.8) + ")";
                ctx.fillRect(0, h * 0.6, w, h * 0.4);

                // Visualiser Rensdyr (simpelt som brune cirkler, spredt over bundområdet)
                ctx.fillStyle = "#d35400";
                let renderCount = Math.min(this.reindeer, 300); // Cap visualisering for ydeevne
                
                // Brug et pseudo-tilfældigt mønster baseret på antal for at undgå flimmer
                for(let i=0; i<renderCount; i++) {
                    // Matematisk deterministisk position baseret på index for statisk rendering, 
                    // men tilføj lidt bølge for at give liv.
                    let bx = (i * 137.5) % w;
                    let by = h * 0.65 + ((i * 93.1) % (h * 0.3));
                    
                    // Mikro-bevægelse
                    bx += Math.sin(this.timer * 10 + i) * 2;

                    ctx.beginPath();
                    ctx.arc(bx, by, 3, 0, Math.PI * 2);
                    ctx.fill();
                }

                // Tegn snefnug for atmosfære
                ctx.fillStyle = "rgba(255,255,255,0.6)";
                for(let i=0; i<this.snowflakes.length; i++) {
                    ctx.beginPath();
                    ctx.arc(this.snowflakes[i].x * w, this.snowflakes[i].y * h, 2, 0, Math.PI*2);
                    ctx.fill();
                }

                // Labels
                ctx.fillStyle = "rgba(255,255,255,0.8)";
                ctx.font = "bold 20px Arial";
                ctx.textAlign = "left";
                ctx.fillText("Rensdyr population (🦌): " + Math.floor(this.reindeer), 20, 30);
                ctx.fillText("Fødegrundlag (🌱): " + Math.floor(this.food), 20, 60);

                this.graph.draw();
            }
        }


        /* =========================================================================
           SIMULATION 3: MARKEDSDYNAMIK
           ========================================================================= */
        class MarketSimulation {
            constructor() {
                this.canvas = getEl("canvas-market");
                this.ctx = this.canvas.getContext("2d");
                this.graph = new GraphEngine("graph-market", ["#e74c3c", "#f39c12"], 200); // Rød=Udbud, Orange=Pris
                
                this.supply = 100; // Enheder af tomater/ærter/jordbær
                this.baseDemand = 100;
                this.price = 15.0; // kr.
                this.totalIncome = 0;
                
                this.timer = 0;
                this.resize();

                // Setup knapper
                getEl("btn-good-harvest").addEventListener("click", () => { 
                    this.supply += 100; // Dobbelt op
                });
                getEl("btn-reset-market").addEventListener("click", () => {
                    this.supply = 100;
                    this.price = 15.0;
                    this.totalIncome = 0;
                    this.graph.data = [];
                });
            }

            resize() {
                const rect = this.canvas.parentElement.getBoundingClientRect();
                this.canvas.width = rect.width;
                this.canvas.height = rect.height;
                this.graph.resize();
            }

            update(dt) {
                this.timer += dt * globalTimeScale;
                
                if (this.timer > 0.1) {
                    this.timer = 0;
                    
                    // Markedslogik
                    // Ligevægtspris er der hvor Udbud = Efterspørgsel.
                    // Antagelse: Hvis udbud er højt, falder prisen. Hvis udbud er lavt, stiger prisen.
                    let targetPrice = 15.0 * (this.baseDemand / Math.max(1, this.supply));
                    
                    // Pris tilpasser sig gradvist (markedstræghed)
                    this.price += (targetPrice - this.price) * 0.1;

                    // Forbrug (Folk køber varer baseret på prisen)
                    let consumed = Math.floor(this.baseDemand / this.price * 2);
                    if (consumed > this.supply) consumed = this.supply;
                    
                    this.supply -= consumed;
                    this.totalIncome += (consumed * this.price);

                    // Bønder producerer lidt hele tiden for at opretholde balance
                    if (this.supply < 100) {
                        this.supply += 2; // Naturlig vækst i sæsonen
                    }

                    this.graph.addData([this.supply, this.price * 10]); // Pris skaleret for graf-læsbarhed
                }

                // UI opdatering
                getEl("stat-supply").innerText = formatDK(this.supply);
                getEl("stat-price").innerText = formatDK(this.price, 2) + " kr.";
                getEl("stat-income").innerText = formatDK(this.totalIncome) + " kr.";
            }

            draw() {
                const ctx = this.ctx;
                const w = this.canvas.width;
                const h = this.canvas.height;

                ctx.clearRect(0, 0, w, h);

                // Markedsbaggrund
                ctx.fillStyle = "#8e44ad";
                ctx.fillRect(0, 0, w, h);
                
                ctx.fillStyle = "#732d91";
                ctx.fillRect(0, h * 0.7, w, h * 0.3);

                // Tegn "Højbede" (Raised beds) fra brugerens kontekst
                ctx.fillStyle = "#8b4513";
                ctx.fillRect(w * 0.2, h * 0.6, w * 0.6, h * 0.2);
                ctx.fillStyle = "#654321";
                ctx.fillRect(w * 0.22, h * 0.62, w * 0.56, h * 0.16);

                // Tegn varer i højbedet (Tomater, ærter, jordbær)
                for(let i=0; i<Math.min(this.supply, 150); i++) {
                    let rx = w * 0.25 + (i * 15) % (w * 0.5);
                    let ry = h * 0.65 + ((i * 7) % (h * 0.1));
                    
                    // Varier farver for tomat/ærter/jordbær
                    let col = i % 3 === 0 ? "#e74c3c" : (i % 3 === 1 ? "#2ecc71" : "#c0392b");
                    ctx.fillStyle = col;
                    ctx.beginPath();
                    ctx.arc(rx, ry, 6, 0, Math.PI*2);
                    ctx.fill();
                }

                // Viser abstrakt vægt for udbud/efterspørgsel
                ctx.strokeStyle = "#fff";
                ctx.lineWidth = 4;
                ctx.beginPath();
                ctx.moveTo(w * 0.5, h * 0.2);
                ctx.lineTo(w * 0.5, h * 0.4);
                ctx.stroke();

                // Vippearm
                let angle = (this.price - 15.0) * -0.05; // Vipper baseret på afvigelse fra standardpris
                if (angle > 0.5) angle = 0.5;
                if (angle < -0.5) angle = -0.5;

                ctx.save();
                ctx.translate(w * 0.5, h * 0.3);
                ctx.rotate(angle);
                
                ctx.beginPath();
                ctx.moveTo(-100, 0);
                ctx.lineTo(100, 0);
                ctx.stroke();

                // Skåle
                ctx.fillStyle = "rgba(255,255,255,0.2)";
                ctx.beginPath();
                ctx.arc(-100, 30, 40, 0, Math.PI);
                ctx.fill();
                ctx.stroke();

                ctx.beginPath();
                ctx.arc(100, 30, 40, 0, Math.PI);
                ctx.fill();
                ctx.stroke();
                
                ctx.restore();

                ctx.fillStyle = "white";
                ctx.font = "14px Arial";
                ctx.textAlign = "center";
                ctx.fillText("Udbud (" + Math.floor(this.supply) + ")", w * 0.5 - 100, h * 0.45);
                ctx.fillText("Pris (" + formatDK(this.price, 2) + " kr)", w * 0.5 + 100, h * 0.45);

                this.graph.draw();
            }
        }

        /* =========================================================================
           HOVED-LOOP OG INITIALISERING
           ========================================================================= */
        const simBridge = new BridgeSimulation();
        const simNature = new NatureSimulation();
        const simMarket = new MarketSimulation();

        let lastTime = 0;

        function gameLoop(timestamp) {
            let dt = (timestamp - lastTime) / 1000;
            if (dt > 0.1) dt = 0.1; // Cap dt for smoothhed ved lag
            lastTime = timestamp;

            // Opdater kun den aktive fane
            if (activeTab === "tab-bridge") {
                simBridge.update(dt);
                simBridge.draw();
            } else if (activeTab === "tab-nature") {
                simNature.update(dt);
                simNature.draw();
            } else if (activeTab === "tab-market") {
                simMarket.update(dt);
                simMarket.draw();
            }

            requestAnimationFrame(gameLoop);
        }

        // Håndter vindues-resize
        window.addEventListener('resize', function() {
            simBridge.resize();
            simNature.resize();
            simMarket.resize();
        });

        // Start animationen
        requestAnimationFrame(function(timestamp) {
            lastTime = timestamp;
            gameLoop(timestamp);
        });

    </script>
